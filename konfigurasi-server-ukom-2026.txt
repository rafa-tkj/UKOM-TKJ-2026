    25. KONFIGURASI OPSI BIND9

    Buka file:

    sudo nano /etc/bind/named.conf.options

    Sesuaikan konfigurasi menjadi:

    options {
    directory "/var/cache/bind";

    listen-on {
        127.0.0.1;
        100.100.20.2;
    };

    listen-on-v6 { none; };

    allow-query {
        localhost;
        10.10.20.0/28;
        192.168.20.0/24;
    };

    recursion yes;

    allow-recursion {
        localhost;
        10.10.20.0/28;
        192.168.20.0/24;
    };

    allow-query-cache {
        localhost;
        10.10.20.0/28;
        192.168.20.0/24;
    };

    forwarders {
        8.8.8.8;
        1.1.1.1;
    };

    dnssec-validation auto;
};
